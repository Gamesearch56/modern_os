# Fonts, color engine, GTK/Qt theming support.
awww                    # wallpaper daemon with animated transitions — see docs/WIDGETS.md
inter-font              # default interface typeface (docs/DESIGN.md)
ttf-jetbrains-mono-nerd # default monospace typeface, with icon glyphs for the bar/cheatsheet
ttf-liberation          # metric-compatible fallback for documents expecting Arial/Times
noto-fonts              # broad Unicode coverage fallback
noto-fonts-cjk          # CJK fallback (kept from the audited system)
noto-fonts-emoji        # emoji rendering everywhere, including the launcher's `:` emoji mode
qt6-svg                 # SVG decode for Papirus/Adwaita icons in Quickshell IconImage
papirus-icon-theme      # gtk-icon-theme-name / qt6ct / config.json design.iconTheme
breeze                  # Dolphin widget style (modern-launch QT_STYLE_OVERRIDE); ColorScheme=modern, no QSS
adwaita-icon-theme      # fallback names when Papirus has no match
hicolor-icon-theme      # freedesktop icon fallback
gnome-themes-extra      # provides Adwaita GTK assets some apps fall back to
sassc                   # compiles the generated gtk.css theme templates
imagemagick             # wallpaper-sync.sh blur + boot-downscale variants
