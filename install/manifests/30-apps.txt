# Daily-driver apps. This is the M0 bar: minimal, but enough to be
# usable day to day. Store lists these as removable apps, not system
# packages (scripts/pkg-manifest-names.sh --core).
fastfetch                # terminal banner, themed via matugen (docs/DECISIONS.md, .bashrc)
ffmpeg                    # mpv_thumbnail_script stills (prefer_mpv=no)
ffmpegthumbnailer         # Nautilus video thumbnails
gst-plugins-base          # GStreamer codecs for mpv/browser media
gst-plugins-good
gst-plugins-bad
gst-plugins-ugly
gst-libav
libva-utils               # VA-API diagnostics for hardware video decode
neovim                    # $EDITOR
btop                      # terminal system monitor, complements the bar's performance widget
spotify-launcher          # Spotify client — MPRIS source for the media widget
steam                     # primary game platform
gamemode                  # CPU governor while a game runs
gamescope                 # per-game resolution and upscaling
umu-launcher              # Proton outside Steam: modern battlenet runs and installs Battle.net with it
