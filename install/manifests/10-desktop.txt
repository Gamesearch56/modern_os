# Hyprland desktop core — every profile past 00-base installs this.
hyprland                       # the compositor
mesa                           # OpenGL/EGL (Hyprland also pulls this; listed so m0 is explicit)
greetd                         # login (enabled only by provision-iso.sh, never install.sh)
xdg-user-dirs                  # Desktop/Pictures before wallpaper copy
gvfs                           # Nautilus trash/network mounts
gvfs-mtp                       # phone MTP in Nautilus
hypridle                       # idle handling (screen off, suspend) — see Services/Power
hyprland-qt-support             # org.hyprland.style QML chrome (application-style.conf); polkit UI is phylax-git (10-desktop.aur.txt)
xdg-desktop-portal-hyprland     # screen share / screenshot portal backend
xdg-desktop-portal-gtk          # fallback portal backend (file chooser when KDE one is absent)
quickshell                      # the shell itself — bar, panels, launcher, lock, greeter
qt6-wayland                     # native Wayland for Qt6 apps (Quickshell, Dolphin)
qt6-webengine                   # wallpaper provider: separate qml6 + WebEngine process (scripts/wallpaper-provider.sh)
qt6ct                           # Qt6 theme configuration, written by theme-apply.sh
networkmanager                  # network management daemon
bluez                            # bluetoothd + bluetooth.service; org.bluez for Services/Bluetooth
bluez-utils                      # bluetoothctl and friends (does not depend on bluez)
kdeconnect                       # Android integration — see Services/KdeConnect
pipewire                         # audio/video server
pipewire-alsa                    # ALSA compatibility shim
pipewire-jack                    # JACK compatibility shim (audio production tools)
pipewire-pulse                   # PulseAudio compatibility shim (most apps still expect this)
wireplumber                      # PipeWire session manager, spoken to by Services/Audio
wl-clipboard                     # wl-copy/wl-paste for screenshots, file preview, copy
grim                             # screenshot capture backend
slurp                            # region selection for screenshots
swappy                           # screenshot annotation (GTK, follows token gtk.css)
wf-recorder                      # screen recording backend (Services/Recording)
ddcutil                         # DDC/CI brightness for external displays — see Services/Brightness
brightnessctl                    # XF86 brightness keys in modules/keybinds.lua
kdialog                          # Qt file picker (wallpaper Add; pick-directory.sh)
hyprpicker                       # color picker widget backend
inotify-tools                    # wallpaper-provider.sh watches the download folder
gamemode                         # CPU governor while a game runs — started from autostart.lua
lib32-gamemode                   # 32-bit GameMode hook for older titles
